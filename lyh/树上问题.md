## 树链剖分

struct HLD {
    int n;
    vector<vector<int>> adj;
    vector<int> dep, siz, parent, in, out, ord, top;
    int cur;

    HLD(int n) {
        init(n);
    }
    void init(int n) {
        this->n = n;
        adj.assign(n, {});
        dep.resize(n);
        siz.resize(n);
        parent.resize(n);
        in.resize(n);
        out.resize(n);
        ord.resize(n);
        top.resize(n);
        cur = 0;
    }
    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    void work(int root) {
        top[root] = root;
        dep[root] = 1;
        parent[root] = 0;
        dfs1(root, 0);
        dfs2(root, 0);
    }
    void dfs1(int u, int pa) {
        siz[u] = 1;
        for (auto &v : adj[u]) {
            if (v == pa) {
                continue;
            }
            parent[v] = u;
            dep[v] = dep[u] + 1;
            dfs1(v, u);
            siz[u] += siz[v];
            if (siz[v] > siz[adj[u][0]]) {
                swap(v, adj[u][0]);
            }
        }
    }
    void dfs2(int u, int pa) {
        in[u] = ++cur;
        ord[in[u]] = u;
        for (auto v : adj[u]) {
            if (v == pa) {
                continue;
            }
            top[v] = (v == adj[u][0] ? top[u] : v);
            dfs2(v, u);
        }
        out[u] = cur;
    }
    int lca(int u, int v) {
        while (top[u] != top[v]) {
            if (dep[top[u]] > dep[top[v]]) {
                swap(u, v);
            }
            v = parent[top[v]];
        }
        return dep[u] > dep[v] ? v : u;
    }
};
