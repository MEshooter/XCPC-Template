```cpp
struct LazySegmentTree {
    int n;
    vector<Info> info;
    vector<Tag> tag;
    LazySegmentTree(int n_, Info v_ = Info()) {
        init(n_, v_);
    }
    LazySegmentTree(const vector<Info> &init_) {
        init(init_);
    }
    void init(int n_, Info v_ = Info()) {
        init(vector(n_, v_)); // seg(n + 1)
    }
    void init(const vector<Info> init_) {
        n = init_.size() - 1; // init_ 大小为 n + 1
        info.assign(4 << __lg(n), Info());
        tag.assign(4 << __lg(n), Tag());
        function<void(int, int, int)> build = [&](int p, int l, int r) {
            if (l == r) {
                info[p] = init_[l];
                info[p].l = l;
                info[p].r = r;

                return;
            }
            int m = l + r >> 1;
            build(p << 1, l, m);
            build(p << 1 | 1, m + 1, r);
            pull(p);
        };
        build(1, 1, n);
    }
    void pull(int p) {
        info[p] = info[p << 1] + info[p << 1 | 1];
    }
    void apply(int p, int l, int r, const Tag &v) {
        info[p].apply(v);
        tag[p].apply(v);
    }
    void push(int p, int l, int r) {
        int m = l + r >> 1;
        apply(p << 1, l, m, tag[p]);
        apply(p << 1 | 1, m + 1, r, tag[p]);
        tag[p] = Tag();
    }
    Info rangeQuery(int p, int l, int r, int x, int y) {
        if (x <= l && r <= y) {
            return info[p];
        }
        push(p, l, r);
        int m = l + r >> 1;
        if (y <= m) {
            return rangeQuery(p << 1, l, m, x, y);
        } else if (x >= m + 1) {
            return rangeQuery(p << 1 | 1, m + 1, r, x, y);
        } else {
            return rangeQuery(p << 1, l, m, x, y) + rangeQuery(p << 1 | 1, m + 1, r, x, y);
        }
    }
    Info rangeQuery(int l, int r) {
        return rangeQuery(1, 1, n, l, r);
    }
    void rangeModify(int p, int l, int r, int x, int y, const Tag &v) {
        if (x <= l && r <= y) {
            apply(p, l, r, v);
            return;
        }
        push(p, l, r);
        int m = l + r >> 1;
        if (y <= m) {
            rangeModify(p << 1, l, m, x, y, v);
        } else if (x >= m + 1) {
            rangeModify(p << 1 | 1, m + 1, r, x, y, v);
        } else {
            rangeModify(p << 1, l, m, x, y, v);
            rangeModify(p << 1 | 1, m + 1, r, x, y, v);
        }
        pull(p);
    }
    void rangeModify(int l, int r, const Tag &v) {
        rangeModify(1, 1, n, l, r, v);
    }

    void modify(int p, int l, int r, int x, const Info &v) {
        if (l == r) {
            info[p] = v;
            return;
        }
        push(p, l, r);
        int m = l + r >> 1;
        if (x <= m) {
            modify(p << 1, l, m, x, v);
        } else {
            modify(p << 1 | 1, m + 1, r, x, v);
        }
        pull(p);
    }
    void modify(int x, const Info &v) {
        modify(1, 1, n, x, v);
    }    
    
    template<class F>
    int findFirst(int p, int l, int r, int x, int y, F &&pred) {
        if (y < l || r < x) {
            return -1;
        }
        if (x <= l && r <= y && !pred(info[p])) {
            return -1;
        }
        if (l == r) {
            return l;
        }
        push(p, l, r);
        int m = l + r >> 1;
        int res = findFirst(p << 1, l, m, x, y, pred);
        if (res == -1) {
            res = findFirst(p << 1 | 1, m + 1, r, x, y, pred);
        }
        return res;
    }
    template<class F>
    int findFirst(int l, int r, F &&pred) {
        return findFirst(1, 1, n, l, r, pred);
    }
    template<class F>
    int findLast(int p, int l, int r, int x, int y, F &&pred) {
        if (y < l || r < x) {
            return -1;
        }
        if (x <= l && r <= y && !pred(info[p])) {
            return -1;
        }
        if (l == r) {
            return l;
        }
        push(p, l, r);
        int m = l + r >> 1;
        int res = findLast(p << 1 | 1, m + 1, r, x, y, pred);
        if (res == -1) {
            res = findLast(p << 1, l, m, x, y, pred);
        }
        return res;
    }
    template<class F>
    int findLast(int l, int r, F &&pred) {
        return findLast(1, 1, n, l, r, pred);
    }
};
```

```cpp
// 计数时常令 a[pl[i]] <= a[i]，a[i] > a[pr[i]]，而不是两边都严格小于。这样不会对相同的a[i]计算重复。
vector<int> pl(n + 1, 0), pr(n + 1, n + 1);

    while (!stk.empty() && a[stk.back()] > a[i]) {
        // a[i] 是 a[stk.back()] 右边第一个 < a[stk.back()] 的数

    if (!stk.empty()) {
        pl[i] = stk.back();
    }
    stk.push_back(i);
```

## 线性基
```cpp
struct LinearBasis {
    bool zero;
    vector<int> f, g;
    LinearBasis() {
        init();
    }
    void init() {
        zero = false;
        f.assign(60, 0);
        g.clear();
    }
    bool insert(int x) {
        for (int i = 59; i >= 0; i--) {
            if (x >> i & 1) {
                if (!f[i]) {
                    f[i] = x;
                    return true;
                }
                x ^= f[i];
            }
        }
        zero = true;
        return false;
    }
    int queryMax() {
        int mx = 0;
        for (int i = 59; i >= 0; i--) {
            mx = max(mx, mx ^ f[i]);
        }
        return mx;
    }
    int queryMin() {
        if (zero) {
            return 0;
        }
        for (int i = 0; i <= 59; i++) {
            if (f[i]) {
                return f[i];
            }
        }
    }
    void initKthMax() {
        g.clear();
        for (int i = 0; i <= 59; i++) {
            for (int j = i - 1; j >= 0; j--) {
                if (f[i] >> j & 1) {
                    f[i] ^= f[j];
                }
            }
            if (f[i]) {
                g.push_back(f[i]);
            }
        }
    }
    int queryKthMax(int k) {
        int tot = g.size(), res = 0;
        if (k <= 0 || k > (1LL << tot)) {
            return -1;
        }
        k -= zero;
        for (int i = 0; i < tot; i++) {
            if (k >> i & 1) {
                res ^= g[i];
            }
        }
        return res;
    }
};
```

## ST表
```cpp
template<class T>
struct SparseTable {
    using Func = function<T(T, T)>; // lambda
    int n;
    vector<int> log2;
    vector<vector<T>> st;
    Func func;

    SparseTable(const vector<T> &a, Func f) {
        init(a, f);
    }
    void init(const vector<T> &a, Func f) {
        n = a.size();
        func = f;
        int k = __lg(n) + 1;
        log2.assign(n, 0);
        st.assign(n, vector<T>(k));
        for (int i = 2; i < n; i++) {
            log2[i] = log2[i >> 1] + 1;
        }
        for (int i = 1; i < n; i++) {
            st[i][0] = a[i];
        }
        for (int j = 1; j < k; j++) {
            for (int i = 1; i + (1 << j) - 1 < n; i++) {
                st[i][j] = func(st[i][j - 1], st[i + (1 << j - 1)][j - 1]);
            }
        }
    }
    T query(int l, int r) {
        int k = log2[r - l + 1];
        return func(st[l][k], st[r - (1 << k) + 1][k]);
    }
};
```

## 异或哈希
```cpp
mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
struct XorHash {
    int n;
    vector<int> Base, Xor, Hash;
    XorHash(const vector<int> &a) {
        init(a);
    }
    void init(const vector<int> &a) {
        n = a.size();
        Base.assign(n, 0);
        Xor.assign(n, 0);
        Hash.assign(n, 0);
        for (int i = 1; i < n; i++) {
            Base[i] = rng();
            Xor[i] = Xor[i - 1] ^ Base[i];
        }
        for (int i = 1; i < n; i++) {
            Hash[i] = Hash[i - 1] ^ Base[a[i]];
        }
    }
    bool isPemutation(int l, int r, int s, int t) { // a[l, r]是否是整数[s, t]的排列
        return (Hash[r] ^ Hash[l - 1]) == (Xor[t] ^ Xor[s - 1]);
    }
};
```

```cpp
        } else if (o == 2) {
            // 删除
            auto it = t.lower_bound({x, 0});
            if (it != t.end() && (*it)[0] == x) {
                t.erase(it);
            }
        } else if (o == 3) {
            // 排名
            cout << t.order_of_key({x, 0}) + 1 << "\n"; 
        } else if (o == 4) {
            // 第 x 大，find_by_order 是 0-based，要查 x - 1
            cout << (*t.find_by_order(x - 1))[0] << "\n";
        } else if (o == 5) {
            // 前驱
            cout << (*--t.lower_bound({x, 0}))[0] << "\n";
        } else {
            // 后继
            cout << (*t.upper_bound({x, idx}))[0] << "\n";
        }
    }
}
```

## pbds_pq
```cpp
using pbds_pq = __gnu_pbds::priority_queue<array<int, 2>, greater<>>;

void solve() {
    int n, m;
    cin >> n >> m;
    
    vector<pbds_pq> pq(n + 1);
    for (int i = 1; i <= n; i++) {
        int x;
        cin >> x;
        pq[i].push({x, i});
    }
    
    vector<bool> del(n + 1);
    DSU dsu(n + 1);

    while (m--) {
        int o;
        cin >> o;
        
        if (o == 1) {
            // 合并第 x 个数和第 y 个数所在的堆
            int x, y;
            cin >> x >> y;
            
            int fx = dsu.find(x);
            int fy = dsu.find(y);
            if (del[x] || del[y] || fx == fy) {
                continue;
            }
            if (pq[fx].size() < pq[fy].size()) {
                swap(fx, fy);
            }
            pq[fx].join(pq[fy]);
            dsu.merge(fx, fy);

        } else {
            // 删除第 x 个数所在堆的堆顶
            int x;
            cin >> x;

            if (del[x]) {
                cout << "-1\n";
            } else {
                x = dsu.find(x);

                auto [val, id] = pq[x].top();
                pq[x].pop();

                del[id] = true;
                cout << val << "\n";
            }
        }
    }
}
```

## 树状数组套线段树
```cpp
int Y = 2e5;
struct Node {
    int l = 0, r = 0;
    int ls = 0, rs = 0;
    int sum = 0, add = 0;
};
vector<Node> in;
int newNode(int l, int r) {
    Node node;
    node.l = l;
    node.r = r;
    in.push_back(node);
    return in.size() - 1;
}
void push(int p) {
    if (in[p].add == 0 || in[p].l == in[p].r) {
        return;
    }
    int m = (in[p].l + in[p].r) >> 1;
    if (!in[p].ls) {
        in[p].ls = newNode(in[p].l, m);
    }
    if (!in[p].rs) {
        in[p].rs = newNode(m + 1, in[p].r);
    }
    in[in[p].ls].sum += (m - in[p].l + 1) * in[p].add;
    in[in[p].ls].add += in[p].add;
    in[in[p].rs].sum += (in[p].r - m) * in[p].add;
    in[in[p].rs].add += in[p].add;
    in[p].add = 0;
}
void pull(int p) {
    in[p].sum = 0;
    if (in[p].ls) {
        in[p].sum += in[in[p].ls].sum;
    }
    if (in[p].rs) {
        in[p].sum += in[in[p].rs].sum;
    }
}
int rangeModify(int p, int l, int r, int yl, int yr, int k) {
    if (yr < l || r < yl) {
        return p;
    }
    if (!p) {
        p = newNode(l, r);
    }
    if (yl <= l && r <= yr) {
        in[p].sum += (r - l + 1) * k;
        in[p].add += k;
        return p;
    }
    push(p);
    int m = (l + r) >> 1;
    in[p].ls = rangeModify(in[p].ls, l, m, yl, yr, k);
    in[p].rs = rangeModify(in[p].rs, m + 1, r, yl, yr, k);
    pull(p);
    return p;
}
int rangeQuery(int p, int yl, int yr) {
    if (!p || yr < in[p].l || in[p].r < yl) {
        return 0;
    }
    if (yl <= in[p].l && in[p].r <= yr) {
        return in[p].sum;
    }
    push(p);
    return rangeQuery(in[p].ls, yl, yr) + rangeQuery(in[p].rs, yl, yr);
}

int X = 2e5;
struct OutNode {
    int inRoot = 0;
};
vector<OutNode> out;
void add(int x, int yl, int yr, int k) {
    assert(x > 0);
    for (; x <= X; x += x & -x) {
        out[x].inRoot = rangeModify(out[x].inRoot, 1, Y, yl, yr, k);
    }
}
int pre(int x, int yl, int yr) {
    assert(x <= X);
    int res = 0;
    for (; x; x -= x & -x) {
        res += rangeQuery(out[x].inRoot, yl, yr);
    }
    return res;
}
```
