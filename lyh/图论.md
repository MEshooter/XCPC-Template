## SCC

```cpp
struct SCC {
    int n;
    vector<vector<int>> adj;
    vector<int> stk;
    vector<int> dfn, low, bel;
    int cur, cnt;
    
    SCC(int n) {
        init(n);
    }
    void init(int n) {
        this->n = n;
        adj.assign(n, {});
        dfn.assign(n, 0);
        low.resize(n);
        bel.assign(n, 0);
        stk.clear();
        cur = cnt = 0;
    }
    void addEdge(int u, int v) {
        adj[u].push_back(v);
    }
    void Tarjan(int u) {
        dfn[u] = low[u] = ++cur;
        stk.push_back(u);
        
        for (auto v : adj[u]) {
            if (!dfn[v]) {
                Tarjan(v);
                low[u] = min(low[u], low[v]);
            } else if (!bel[v]) {
                low[u] = min(low[u], dfn[v]);
            }
        }
        
        if (dfn[u] == low[u]) {
            cnt++;
            int v;
            do {
                v = stk.back();
                stk.pop_back();
                bel[v] = cnt;
            } while (v != u);
        }
    }
    vector<int> work() {
        for (int i = 1; i < n; i++) {
            if (!dfn[i]) {
                Tarjan(i);
            }
        }
        return bel;
    }
};
```

```cpp
struct VBCC {
    int n;
    vector<vector<int>> adj;
    vector<int> stk;
    vector<int> dfn, low;
    int cur, cnt;

    VBCC(int n) {
        init(n);
    }
    void init(int n) {
        this->n = n;
        adj.assign(n, {});
        dfn.assign(n, 0);
        low.resize(n);
        cur = cnt = 0;
    }
    void addEdge(int u, int v) {
        adj[u].push_back(v);
        adj[v].push_back(u);
    }
    void Tarjan(int u, int pa) {
        dfn[u] = low[u] = ++cur;
        stk.push_back(u);
        
        for (auto v : adj[u]) {
            if (v == pa) {
                continue;
            }
            if (!dfn[v]) {
                Tarjan(v, u);
                low[u] = min(low[u], low[v]);

                if (low[v] >= dfn[u]) {
                    vbcc.emplace_back();
                    cnt++;
                    int y;
                    do {
                        y = stk.back();
                        stk.pop_back();
                        vbcc[cnt].push_back(y);
                    } while (y != v);
                    vbcc[cnt].push_back(u);
                }
            } else {
                low[u] = min(low[u], dfn[v]);
            }
        }

        if (pa == 0 && adj[u].size() == 0) {
            vbcc.emplace_back();
            cnt++;
            vbcc[cnt].push_back(u);
        }
    }
    auto work() {
        for (int i = 1; i < n; i++) {
            if (!dfn[i]) {
                Tarjan(i, 0);
            }
        }
        return vbcc;
    }
};
```

## 最大流
```cpp
template<class T>
struct MaxFlow {
    struct _Edge {
        int to;
        T cap;
    };
    int n;
    vector<_Edge> e;
    vector<vector<int>> g;
    vector<int> cur, h;
    MaxFlow(int n) {
        init(n);
    }
    void init(int n) {
        this->n = n;
        e.clear();
        g.assign(n, {});
        cur.resize(n);
        h.resize(n);
    }
    bool bfs(int s, int t) {
        h.assign(n, -1);
        queue<int> q;
        h[s] = 0;
        q.push(s);
        while (!q.empty()) {
            auto u = q.front();
            q.pop();
            for (auto i : g[u]) {
                auto [v, c] = e[i];
                if (c > 0 && h[v] == -1) {
                    h[v] = h[u] + 1;
                    if (v == t) {
                        return true;
                    }
                    q.push(v);
                }
            }
        }
        return false;
    }
    T dfs(int u, int t, T f) {
        if (u == t) {
            return f;
        }
        auto r = f;
        for (int &i = cur[u]; i < int(g[u].size()); i++) {
            auto j = g[u][i];
            auto [v, c] = e[j];
            if (c > 0 && h[v] == h[u] + 1) {
                auto a = dfs(v, t, min(r, c));
                e[j].cap -= a;
                e[j ^ 1].cap += a;
                r -= a;
                if (r == 0) {
                    return f;
                }
            }
        }
        return f - r;
    }
    void addEdge(int u, int v, T cap) {
        g[u].push_back(e.size());
        e.push_back({v, cap});
        g[v].push_back(e.size());
        e.push_back({u, 0});
    }
    T flow(int s, int t) {
        T ans = 0;
        while (bfs(s, t)) {
            cur.assign(n, 0);
            ans += dfs(s, t, inf);
        }
        return ans;
    }
    vector<bool> minCut() {
        vector<bool> c(n);
        for (int i = 0; i < n; i++) {
            c[i] = (h[i] != -1);
        }
        return c;
    }
    struct Edge {
        int from;
        int to;
        T cap;
        T flow;
    };
    vector<Edge> edges() {
        vector<Edge> res;
        for (int i = 0; i < e.size(); i += 2) {
            res.push_back({e[i + 1].to, e[i].to, e[i].cap + e[i + 1].cap, e[i + 1].cap});
        }
        return res;
    }
};
```

## 最小费用最大流
```cpp
template<class T>
struct MinCostFlow {
    struct _Edge {
        int to;
        T cap;
        T cost;
    };
    int n;
    vector<_Edge> e;
    vector<vector<int>> g;
    vector<T> h, dis;
    vector<int> pre;
    bool Dijkstra(int s, int t) {
        dis.assign(n, inf);
        pre.assign(n, -1);
        priority_queue<pair<T, int>, vector<pair<T, int>>, greater<pair<T, int>>> q;
        dis[s] = 0;
        q.push({0, s});
        while (!q.empty()) {
            auto [d, u] = q.top();
            q.pop();
            if (dis[u] < d) {
                continue;
            }
            for (auto id : g[u]) {
                auto [v, cap, cost] = e[id];
                if (cap > 0 && dis[v] > d + h[u] - h[v] + cost) {
                    dis[v] = d + h[u] - h[v] + cost;
                    pre[v] = id;
                    q.push({dis[v], v});
                }
            }
        }
        return dis[t] != inf;
    }
    MinCostFlow() {}
    MinCostFlow(int n_) {
        init(n_);
    }
    void init(int n_) {
        n = n_;
        e.clear();
        g.assign(n, {});
    }
    void addEdge(int u, int v, T cap, T cost) {
        g[u].push_back(e.size());
        e.push_back({v, cap, cost});
        g[v].push_back(e.size());
        e.push_back({u, 0, -cost});
    }
    pair<T, T> flow(int s, int t) {
        T flow = 0, cost = 0;
        h.assign(n, 0);
        while (Dijkstra(s, t)) {
            for (int i = 0; i < n; i++) {
                h[i] += dis[i];
            }
            T aug = inf;
            for (int i = t; i != s; i = e[pre[i] ^ 1].to) {
                aug = min(aug, e[pre[i]].cap);
            }
            for (int i = t; i != s; i = e[pre[i] ^ 1].to) {
                e[pre[i]].cap -= aug;
                e[pre[i] ^ 1].cap += aug;
            }
            flow += aug;
            cost += aug * h[t];
        }
        return {flow, cost};
    }
    struct Edge {
        int from;
        int to;
        T cap;
        T cost;
        T flow;
    };
    vector<Edge> edges() {
        vector<Edge> res;
        for (int i = 0; i < e.size(); i += 2) {
            Edge ed = {e[i + 1].to, e[i].to, e[i].cap + e[i + 1].cap, e[i].cost, e[i + 1].cap};
            res.push_back(ed);
        }
        return res;
    }
};
```

## SPFA
```cpp
auto SPFA = [&](int s) {
    vector<int> dis(n + 1, inf), in_q(n + 1), cnt(n + 1);
    dis[s] = 0;
    queue<int> q;
    q.push(s);
    in_q[s] = true;
    while (q.size()) {
        auto u = q.front();
        q.pop();
        in_q[u] = false;
        for (auto [v, w] : e[u]) {
            if (dis[v] > dis[u] + w) {
                dis[v] = dis[u] + w;
                if (!in_q[v]) {
                    q.push(v);
                    in_q[v] = 1;
                    cnt[v] = cnt[u] + 1;
                    if (cnt[v] >= n) {
                        return true;
                    }
                }
            }
        }
    }
    return false;
};
```

## 欧拉路径
```cpp
// 有向图
void solve() {
    int n, m;
    cin >> n >> m;
    vector<vector<int>> e(n + 1);
    vector<int> in(n + 1), out(n + 1);
    for (int i = 1; i <= m; i++) {
        int u, v;
        cin >> u >> v;
        e[u].push_back(v);
        out[u]++;
        in[v]++;
    }
    
    int cnt1 = 0, cnt2 = 0, s = 1;
    for (int i = 1; i <= n; i++) {
        sort(e[i].begin(), e[i].end());
        if (out[i] != in[i]) {
            if (out[i] == in[i] + 1) {
                cnt1++;
                s = i;
            } else if (out[i] + 1 == in[i]) {
                cnt2++;
            } else {
                cout << "No\n";
                return;
            }
        }
    }
    if (!(cnt1 == 1 && cnt2 == 1 || cnt1 == 0 && cnt2 == 0)) {
        cout << "No\n";
        return;
    }

    vector<int> cur(n + 1);
    stack<int> st;
    auto dfs = [&](auto self, int u) -> void {
        for (int i = cur[u]; i < e[u].size(); i = cur[u]) {
            cur[u] = i + 1;
            int v = e[u][i];
            self(self, v);
        }
        st.push(u);
    };
    dfs(dfs, s);

    while (st.size()) {
        cout << st.top() << " ";
        st.pop();
    }
}

// 无向图
void solve() {
    int n = 500, m;
    cin >> m;
    vector<vector<array<int, 2>>> e(n + 1);
    vector<int> d(n + 1);
    int s = n;
    for (int i = 1; i <= m; i++) {
        int u, v;
        cin >> u >> v;
        e[u].push_back({v, i});
        e[v].push_back({u, i});
        d[u]++;
        d[v]++;
        s = min(s, min(u, v));
    }
    
    for (int i = n; i >= 1; i--) {
        sort(e[i].begin(), e[i].end());
        if (d[i] % 2 == 1) {
            s = i;
        }
    }

    vector<bool> vis(m + 1);
    vector<int> cur(n + 1);
    stack<int> st;
    auto dfs = [&](auto &&self, int u) -> void {
        for (int &j = cur[u]; j < e[u].size(); j++) {
            auto [v, i] = e[u][j];
            if (!vis[i]) {
                vis[i] = true;
                self(self, v);
                // ans.push_back(i);
            }
        }
        st.push(u);
    };
    dfs(dfs, s);

    while (st.size()) {
        cout << st.top() << "\n";
        st.pop();
    }
}
```
