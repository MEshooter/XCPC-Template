```cpp
struct Tag {
    int x;
    void apply(const Tag &t) {

    }
};

struct Info {
    int l, r;

    void apply(const Tag &t) {

    }
};
Info operator+(const Info &u, const Info &v) {
    Info res;
    res.l = u.l;
    res.r = v.r;
    
    return res;
}

struct LazySegmentTree {
    int n;
    vector<Info> info;
    vector<Tag> tag;
    LazySegmentTree(int n_, Info v_ = Info()) {
        init(n_, v_);
    }
    LazySegmentTree(const vector<Info> &init_) {
        init(init_);
    }
    void init(int n_, Info v_ = Info()) {
        init(vector(n_, v_)); // seg(n + 1)
    }
    void init(const vector<Info> init_) {
        n = init_.size() - 1; // init_ 大小为 n + 1
        info.assign(4 << __lg(n), Info());
        tag.assign(4 << __lg(n), Tag());
        function<void(int, int, int)> build = [&](int p, int l, int r) {
            if (l == r) {
                info[p] = init_[l];
                info[p].l = l;
                info[p].r = r;

                return;
            }
            int m = l + r >> 1;
            build(p << 1, l, m);
            build(p << 1 | 1, m + 1, r);
            pull(p);
        };
        build(1, 1, n);
    }
    void pull(int p) {
        info[p] = info[p << 1] + info[p << 1 | 1];
    }
    void apply(int p, int l, int r, const Tag &v) {
        info[p].apply(v);
        tag[p].apply(v);
    }
    void push(int p, int l, int r) {
        int m = l + r >> 1;
        apply(p << 1, l, m, tag[p]);
        apply(p << 1 | 1, m + 1, r, tag[p]);
        tag[p] = Tag();
    }
    Info rangeQuery(int p, int l, int r, int x, int y) {
        if (x <= l && r <= y) {
            return info[p];
        }
        push(p, l, r);
        int m = l + r >> 1;
        if (y <= m) {
            return rangeQuery(p << 1, l, m, x, y);
        } else if (x >= m + 1) {
            return rangeQuery(p << 1 | 1, m + 1, r, x, y);
        } else {
            return rangeQuery(p << 1, l, m, x, y) + rangeQuery(p << 1 | 1, m + 1, r, x, y);
        }
    }
    Info rangeQuery(int l, int r) {
        return rangeQuery(1, 1, n, l, r);
    }
    void rangeModify(int p, int l, int r, int x, int y, const Tag &v) {
        if (x <= l && r <= y) {
            apply(p, l, r, v);
            return;
        }
        push(p, l, r);
        int m = l + r >> 1;
        if (y <= m) {
            rangeModify(p << 1, l, m, x, y, v);
        } else if (x >= m + 1) {
            rangeModify(p << 1 | 1, m + 1, r, x, y, v);
        } else {
            rangeModify(p << 1, l, m, x, y, v);
            rangeModify(p << 1 | 1, m + 1, r, x, y, v);
        }
        pull(p);
    }
    void rangeModify(int l, int r, const Tag &v) {
        rangeModify(1, 1, n, l, r, v);
    }

    void modify(int p, int l, int r, int x, const Info &v) {
        if (l == r) {
            info[p] = v;
            return;
        }
        push(p, l, r);
        int m = l + r >> 1;
        if (x <= m) {
            modify(p << 1, l, m, x, v);
        } else {
            modify(p << 1 | 1, m + 1, r, x, v);
        }
        pull(p);
    }
    void modify(int x, const Info &v) {
        modify(1, 1, n, x, v);
    }    
    
    template<class F>
    int findFirst(int p, int l, int r, int x, int y, F &&pred) {
        if (y < l || r < x) {
            return -1;
        }
        if (x <= l && r <= y && !pred(info[p])) {
            return -1;
        }
        if (l == r) {
            return l;
        }
        push(p, l, r);
        int m = l + r >> 1;
        int res = findFirst(p << 1, l, m, x, y, pred);
        if (res == -1) {
            res = findFirst(p << 1 | 1, m + 1, r, x, y, pred);
        }
        return res;
    }
    template<class F>
    int findFirst(int l, int r, F &&pred) {
        return findFirst(1, 1, n, l, r, pred);
    }
    template<class F>
    int findLast(int p, int l, int r, int x, int y, F &&pred) {
        if (y < l || r < x) {
            return -1;
        }
        if (x <= l && r <= y && !pred(info[p])) {
            return -1;
        }
        if (l == r) {
            return l;
        }
        push(p, l, r);
        int m = l + r >> 1;
        int res = findLast(p << 1 | 1, m + 1, r, x, y, pred);
        if (res == -1) {
            res = findLast(p << 1, l, m, x, y, pred);
        }
        return res;
    }
    template<class F>
    int findLast(int l, int r, F &&pred) {
        return findLast(1, 1, n, l, r, pred);
    }
};
```

## 单调栈和单调队列
```cpp
// 计数时常令 a[pl[i]] <= a[i]，a[i] > a[pr[i]]，而不是两边都严格小于。这样不会对相同的a[i]计算重复。
vector<int> pl(n + 1, 0), pr(n + 1, n + 1);
vector<int> stk;
for (int i = 1; i <= n; i++) {
    while (!stk.empty() && a[stk.back()] > a[i]) {
        // a[i] 是 a[stk.back()] 右边第一个 < a[stk.back()] 的数
        pr[stk.back()] = i;
        stk.pop_back();
    }
    // a[stk.back()] 是 a[i] 左边第一个 <= a[i] 的数
    if (!stk.empty()) {
        pl[i] = stk.back();
    }
    stk.push_back(i);
}

deque<int> q;
for (int i = 1; i <= n; i++) {
    while (!q.empty() && q.front() < i - k + 1) {
        q.pop_front();
    }
    while (!q.empty() && a[q.back()] >= a[i]) {
        q.pop_back();
    }
    q.push_back(i);
    if (i >= k) { // 最小值
        cout << a[q.front()] << " \n"[i == n];
    }
}
```

## 线性基
```cpp
struct LinearBasis {
    bool zero;
    vector<int> f, g;
    LinearBasis() {
        init();
    }
    void init() {
        zero = false;
        f.assign(60, 0);
        g.clear();
    }
    bool insert(int x) {
        for (int i = 59; i >= 0; i--) {
            if (x >> i & 1) {
                if (!f[i]) {
                    f[i] = x;
                    return true;
                }
                x ^= f[i];
            }
        }
        zero = true;
        return false;
    }
    int queryMax() {
        int mx = 0;
        for (int i = 59; i >= 0; i--) {
            mx = max(mx, mx ^ f[i]);
        }
        return mx;
    }
    int queryMin() {
        if (zero) {
            return 0;
        }
        for (int i = 0; i <= 59; i++) {
            if (f[i]) {
                return f[i];
            }
        }
    }
    void initKthMax() {
        g.clear();
        for (int i = 0; i <= 59; i++) {
            for (int j = i - 1; j >= 0; j--) {
                if (f[i] >> j & 1) {
                    f[i] ^= f[j];
                }
            }
            if (f[i]) {
                g.push_back(f[i]);
            }
        }
    }
    int queryKthMax(int k) {
        int tot = g.size(), res = 0;
        if (k <= 0 || k > (1LL << tot)) {
            return -1;
        }
        k -= zero;
        for (int i = 0; i < tot; i++) {
            if (k >> i & 1) {
                res ^= g[i];
            }
        }
        return res;
    }
};
```

## ST表
```cpp
template<class T>
struct SparseTable {
    using Func = function<T(T, T)>; // lambda
    int n;
    vector<int> log2;
    vector<vector<T>> st;
    Func func;

    SparseTable(const vector<T> &a, Func f) {
        init(a, f);
    }
    void init(const vector<T> &a, Func f) {
        n = a.size();
        func = f;
        int k = __lg(n) + 1;
        log2.assign(n, 0);
        st.assign(n, vector<T>(k));
        for (int i = 2; i < n; i++) {
            log2[i] = log2[i >> 1] + 1;
        }
        for (int i = 1; i < n; i++) {
            st[i][0] = a[i];
        }
        for (int j = 1; j < k; j++) {
            for (int i = 1; i + (1 << j) - 1 < n; i++) {
                st[i][j] = func(st[i][j - 1], st[i + (1 << j - 1)][j - 1]);
            }
        }
    }
    T query(int l, int r) {
        int k = log2[r - l + 1];
        return func(st[l][k], st[r - (1 << k) + 1][k]);
    }
};
```

## 异或哈希
```cpp
mt19937_64 rng(chrono::steady_clock::now().time_since_epoch().count());
struct XorHash {
    int n;
    vector<int> Base, Xor, Hash;
    XorHash(const vector<int> &a) {
        init(a);
    }
    void init(const vector<int> &a) {
        n = a.size();
        Base.assign(n, 0);
        Xor.assign(n, 0);
        Hash.assign(n, 0);
        for (int i = 1; i < n; i++) {
            Base[i] = rng();
            Xor[i] = Xor[i - 1] ^ Base[i];
        }
        for (int i = 1; i < n; i++) {
            Hash[i] = Hash[i - 1] ^ Base[a[i]];
        }
    }
    bool isPemutation(int l, int r, int s, int t) { // a[l, r]是否是整数[s, t]的排列
        return (Hash[r] ^ Hash[l - 1]) == (Xor[t] ^ Xor[s - 1]);
    }
};
```
