## 组合数
```cpp
int fac[N], inv[N], inv_fac[N];
void binom_init(int n) {
    fac[0] = fac[1] = inv[0] = inv[1] = inv_fac[0] = inv_fac[1] = 1;
    for(int i = 2; i <= n; i++) {
        fac[i] = fac[i - 1] * i % mod;
        inv[i] = (mod - mod / i) * inv[mod % i] % mod;
    }
    for(int i = 2; i <= n; i++) {
        inv_fac[i] = inv_fac[i - 1] * inv[i] % mod;
    }
}
int binom(int n, int m) {
    if (n < 0 || m < 0 || n < m) {
        return 0;
    }
    return fac[n] * inv_fac[m] % mod * inv_fac[n - m] % mod;
}
```

## 线性筛
```cpp
vector<int> primes, minp;
vector<int> phi;
void sieve(int n) {
    minp.assign(n + 1, 0);
    phi.assign(n + 1, 1);
    for (int i = 2; i <= n; i++) {
        if (!minp[i]) {
            primes.push_back(minp[i] = i);
            phi[i] = i - 1;
        }
        for (auto p : primes) {
            if (i * p > n) {
                break;
            }
            minp[i * p] = p;
            if (p == minp[i]) {
                phi[i * p] = phi[i] * p;
                break;
            }
            phi[i * p] = phi[i] * phi[p];
        }
    }
}
```

## NTT
```cpp
constexpr int mod = 998244353;
int g0 = 3, g1;
vector<int> rev;
void NTT(vector<int> &A, bool inverse) {
    int n = A.size();
    if ((int)rev.size() != n) {
        rev.resize(n);
        for (int i = 1; i < n; i++) {
            rev[i] = rev[i >> 1] >> 1 | (i & 1) << (__builtin_ctzll(n) - 1);
        }
    }
    for (int i = 0; i < n; i++) {
        if (i < rev[i]) {
            swap(A[i], A[rev[i]]);
        }
    }
    for (int len = 2; len <= n; len <<= 1) {
        int gn = power(inverse ? g1 : g0, (mod - 1) / len);
        for (int j = 0; j < n; j += len) {
            int g = 1;
            for (int k = j; k < j + len / 2; k++, g = g * gn % mod) {
                int u = A[k], v = g * A[k + len / 2] % mod;
                A[k] = (u + v) % mod;
                A[k + len / 2] = (u - v + mod) % mod;
            }
        }
    }
    if (inverse) {
        int inv = power(n, mod - 2);
        for (int i = 0; i < n; i++) {
            A[i] = A[i] * inv % mod;
        }
    }
}

void solve() {
    g1 = power(g0, mod - 2);
    int n, m;
    cin >> n >> m;
    
    vector<int> a(n + 1), b(m + 1);
    for (int i = 0; i <= n; i++) {
        cin >> a[i];
    }
    for (int i = 0; i <= m; i++) {
        cin >> b[i];
    }
    int len = 1;
    while (len < n + m + 1) {
        len <<= 1;
    }
    a.resize(len);
    b.resize(len);
    NTT(a, 0);
    NTT(b, 0);
    vector<int> c(len);
    for (int i = 0; i < len; i++) {
        c[i] = a[i] * b[i] % mod;
    }
    NTT(c, 1);
    for (int i = 0; i <= n + m; i++) {
        cout << c[i] << " \n"[i == len - 1];
    }
}
```

## FFT
```cpp
using ld = double;
const ld PI = acos(-1);
struct Complex {
    ld x, y;
    Complex(ld x_ = 0, ld y_ = 0) : x(x_), y(y_) {}
    friend Complex operator+(const Complex &u, const Complex &v) {
        return {u.x + v.x, u.y + v.y};
    }
    friend Complex operator-(const Complex &u, const Complex &v) {
        return {u.x - v.x, u.y - v.y};
    }
    friend Complex operator*(const Complex &u, const Complex &v) {
        return {u.x * v.x - u.y * v.y, u.x * v.y + u.y * v.x};
    }
};

vector<int> rev;
void FFT(vector<Complex> &A, bool inverse) {
    int n = A.size();
    if ((int)rev.size() != n) {
        rev.resize(n);
        for (int i = 1; i < n; i++) {
            rev[i] = rev[i >> 1] >> 1 | (i & 1) << (__builtin_ctzll(n) - 1);
        }
    }
    for (int i = 0; i < n; i++) {
        if (i < rev[i]) {
            swap(A[i], A[rev[i]]);
        }
    }
    for (int len = 2; len <= n; len <<= 1) {
        Complex wn(cos(2 * PI / len), sin(2 * PI / len) * (inverse ? -1 : 1));
        for (int j = 0; j < n; j += len) {
            Complex w(1, 0);
            for (int k = j; k < j + len / 2; k++, w = w * wn) {
                Complex u = A[k], v = w * A[k + len / 2];
                A[k] = u + v;
                A[k + len / 2] = u - v;
            }
        }
    }
    if (inverse) {
        for (int i = 0; i < n; i++) {
            A[i].x = (int)(A[i].x / n + 0.5);
        }
    }
}
```

## FWT
```cpp
void Or(int n, int *a, int type) {
    for (int i = 1; i < n; i <<= 1) {
        for (int j = 0; j < n; j += i << 1) {
            for (int k = 0; k < i; k++) {
                (a[i + j + k] += a[j + k] * type + mod) %= mod;
            }
        }
    }
}
void And(int n, int *a, int type) {
    for (int i = 1; i < n; i <<= 1) {
        for (int j = 0; j < n; j += i << 1) {
            for (int k = 0; k < i; k++) {
                (a[j + k] += a[i + j + k] * type + mod) %= mod;
            }
        }
    }
}
void Xor(int n, int *a, int type) {
    for (int i = 1; i < n; i <<= 1) {
        for (int j = 0; j < n; j += i << 1) {
            for (int k = 0; k < i; k++) {
                (a[j + k] += a[i + j + k]) %= mod;
                (a[i + j + k] = a[j + k] - 2 * a[i + j + k] % mod + mod) %= mod;
                (a[j + k] *= type) %= mod;
                (a[i + j + k] *= type) %= mod;
            }
        }
    }
}
void FWT(int n, int *a, int f_type, int t_type) {
    if (f_type == 0) {
        if (t_type == 0) Or(n, a, 1);
        else Or(n, a, -1);
    } else if (f_type == 1) {
        if (t_type == 0) And(n, a, 1);
        else And(n, a, -1);
    } else {
        if (t_type == 0) Xor(n, a, 1);
        else Xor(n, a, power(2, mod - 2));
    }
}
```

## 高斯消元
```cpp
pair<int, vector<ld>> Gauss(vector<vector<ld>> a, vector<ld> b) {
    int n = a.size() - 1, rank = 0;
    for (int j = 1; j <= n; j++) {
        int r = rank + 1;
        for (int i = r + 1; i <= n; i++) {
            if (abs(a[r][j]) < abs(a[i][j])) {
                r = i;
            }
        }
        if (abs(a[r][j]) < eps) {
            continue;
        }
        if (r != rank + 1) {
            swap(a[r], a[rank + 1]);
            swap(b[r], b[rank + 1]);
        }
        auto div = a[rank + 1][j];
        for (int k = j; k <= n; k++) {
            a[rank + 1][k] /= div;
        }
        b[rank + 1] /= div;
        for (int i = 1; i <= n; i++) {
            if (i == rank + 1) {
                continue;
            }
            auto mul = a[i][j];
            for (int k = j; k <= n; k++) {
                a[i][k] -= mul * a[rank + 1][k];
            }
            b[i] -= mul * b[rank + 1];
        }
        rank++;
    }

    if (rank < n) {
        for (int i = rank + 1; i <= n; i++) {
            if (abs(b[i]) > eps) {
                return {0, {}};
            }
        }
        return {2, {}};
    }
    return {1, b};
}
```

## exgcd
```cpp
int exgcd(int a, int b, int &x, int &y) {
    if (b == 0) {
        x = 1, y = 0;
        return a;
    }
    int g = exgcd(b, a % b, y, x);
    y -= a / b * x;
    return g;
}
```

## Lucas
```cpp
int Lucas(int n, int m, int p) {
    if (m == 0) {
        return 1;
    }
    return binom(n % p, m % p, p) * Lucas(n / p, m / p, p) % p;
}
```

```cpp
void solve() {
    int n;
    cin >> n;
    
    int prod = 1;
    vector<int> a(n + 1), b(n + 1);
    for (int i = 1; i <= n; i++) {
        cin >> a[i] >> b[i];
        prod *= a[i];
    }

    int x = 0;
    for (int i = 1; i <= n; i++) {
        int cur = prod / a[i], inv, k;
        // cur * x = 1 - k * a[i]
        exgcd(cur, a[i], inv, k);
        x += (i128)b[i] * cur % prod * inv % prod;
        x %= prod;
    }
    cout << (x + prod) % prod << "\n";
}
```

## 自适应辛普森法
```cpp
ld a, b, c, d, L, R;
ld f(ld x) {
    return (c * x + d) / (a * x + b);
}
ld Simpson(ld l, ld r) {
    ld mid = (l + r) / 2;
    return (r - l) * (f(l) + f(r) + 4 * f(mid)) / 6;
}
ld ASR(ld l, ld r, ld res, ld eps) {
    ld mid = (l + r) / 2;
    ld fl = Simpson(l, mid), fr = Simpson(mid, r);
    if (fabsl(fl + fr - res) < 15 * eps) {
        return fl + fr + (fl + fr - res) / 15;
    }
    return ASR(l, mid, fl, eps / 2) + ASR(mid, r, fr, eps / 2);
}

void solve() {
    cin >> a >> b >> c >> d >> L >> R;
    cout << fixed << setprecision(6) << ASR(L, R, Simpson(L, R), 1e-6);
}
```

```cpp
// O(m) 计算 f(n, m) = \sum_{i = 1}^{n} i^m
int f(int n, int m) {
    vector<int> pre(m + 3, 1), suf(m + 4, 1);
    for (int i = 1; i <= m + 2; i++) {
        pre[i] = pre[i - 1] * (n - i + mod) % mod;
    }
    for (int i = m + 2; i >= 1; i--) {
        suf[i] = suf[i + 1] * (n - i + mod) % mod;
    }
    int res = 0;
    for (int i = 1; i <= m + 2; i++) {
        int u = pre[i - 1] * suf[i + 1] % mod;
        int v = inv_fac[i - 1] * (inv_fac[m + 2 - i] * ((m + 2 - i) % 2 == 1 ? mod - 1 : 1) % mod) % mod;
        res += pre_pw[i][m] * u % mod * v % mod;
        res %= mod;
    }

}
```

## BSGS
```cpp
void solve() {
    int a, b, m;
    cin >> m >> a >> b;

    auto power = [&](int a, int b) {
        int res = 1;
        for (a %= m; b; b >>= 1, a = a * a % m) {
            if (b & 1) {
                res = res * a % m;
            }
        }
        return res;
    };

    unordered_map<int, int> mp;
    int sq = ceil(sqrt(m));
    int pw = 1;
    for (int B = 0; B <= sq; B++) {
        mp[b * pw % m] = B;
        pw = pw * a % m;
    }
    int mn = inf;
    pw = 1;
    int pw_sq = power(a, sq) % m;
    for (int A = 0; A <= sq; A++) {
        if (mp.count(pw)) {
            int B = mp[pw];
            if (A * sq - B >= 0) {
                mn = min(mn, A * sq - B);
            }
        }
        pw = pw * pw_sq % m;
    }
    cout << (mn == inf ? "no solution" : to_string(mn)) << "\n";
}
```
